cmake_minimum_required(VERSION 3.24)
project(CreationMaster26 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(CreationMaster26 WIN32 src/main.cpp src/database_engine.cpp)
target_include_directories(CreationMaster26 PRIVATE src)
target_compile_definitions(CreationMaster26 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CreationMaster26 PRIVATE bcrypt comctl32 comdlg32 shell32 ole32 user32 gdi32)
if(MSVC)
  target_compile_options(CreationMaster26 PRIVATE /W4 /permissive- /EHsc)
  target_link_options(CreationMaster26 PRIVATE /SUBSYSTEM:WINDOWS)
endif()
add_executable(EngineSmokeTest tests/engine_smoke.cpp src/database_engine.cpp)
target_include_directories(EngineSmokeTest PRIVATE src)
target_link_libraries(EngineSmokeTest PRIVATE bcrypt)
